`;
});
}

// KRIDA CRUD OPERATIONS
function updateKrida(id, field, value) {
const krida = appState.kridas.find(k => k.id === id);
if(krida) {
krida[field] = value;
localStorage.setItem('sbk_kridas', JSON.stringify(appState.kridas));
renderKridaCards();
}
}

function deleteKrida(id) {
appState.kridas = appState.kridas.filter(k => k.id !== id);
localStorage.setItem('sbk_kridas', JSON.stringify(appState.kridas));
renderKridaCards();
}

let tempPhotoBase64 = "";
function previewPhotoInput(input) {
if (input.files && input.files[0]) {
const reader = new FileReader();
reader.onload = function(e) {
tempPhotoBase64 = e.target.result;
document.getElementById('foto-preview').src = tempPhotoBase64;

}
reader.readAsDataURL(input.files[0]);
}
}
